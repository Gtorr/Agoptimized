# [ Progress event ]
province_event = {
    id = colony.3
    desc = "ProgressIsMade"
    is_triggered_only = yes	# colony.2 [loop event}
    trigger = {
        has_province_flag = colony_province
    }
    hide_window = yes
    immediate = {
        #name = "Ok"
        #Colony work
        if = {
            limit = {
                has_instances_of_province_modifier = {
                    modifier = colony_work
                    amount = 1
                }
            }
            remove_province_modifiers = {
                modifier = colony_work
                amount = 1
            }
            if = {
                limit = {
                    traditional_nomad_territory_trigger = yes
                    has_instances_of_province_modifier = {
                        modifier = colony_work
                        amount = 1
                    }
                }
                remove_holding_modifiers = {
                    modifier = colony_work
                    amount = 1
                }
            }
            province_event = {
                id = colony.2				# Next timer
            }
        }
        else_if = {
            # Halfway
            limit = {
                has_province_modifier = colony
                #NOT = {
                #	has_province_modifier = partial_colony
                #}
            }
            add_province_modifier = {
                name = colony_work
                stacking = yes
                duration = -1
            }
            add_province_modifier = {
                name = colony_work
                stacking = yes
                duration = -1
            }
            add_province_modifier = {
                name = colony_work
                stacking = yes
                duration = -1
            }
            province_event = {
                id = colony.4				#Partial Colony
            }
            province_event = {
                id = colony.2				# Next timer
            }
        }
        else_if = {
            # End
            limit = {
                has_province_modifier = partial_colony
                NOT = {
                    has_instances_of_province_modifier = {
                        modifier = colony_work
                        amount = 1
                    }
                }
            }
            if = {
                limit = {
                    NOT = {
                        any_province_holding = {
                            holding_type = temple
                        }
                    }
                }
                build_holding = {
                    type = temple
                }
            }
            any_province_holding = {
                limit = {
                    holding_type = temple
                }
                add_building = tp_town_1
            }
            province_event = {
                id = colony.4				#Finish colony
            }
        }
    }
}

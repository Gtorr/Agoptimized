# A stage has been completed
province_event = {
	id = iron_throne.16
	picture = GFX_evt_castle_construction
	border = GFX_event_normal_frame_economy
	is_triggered_only = yes
	desc = {
		text = EVTDESCAiron_throne.16
		trigger = {
			has_province_modifier = summerhall_stage1
		}
	}
	desc = {
		text = EVTDESCBiron_throne.16
		trigger = {
			has_province_modifier = summerhall_stage2
		}
	}
	desc = {
		text = EVTDESCCiron_throne.16
		trigger = {
			has_province_modifier = summerhall_stage3
		}
	}
	immediate = {
		if = {
			limit = {
				has_province_modifier = summerhall_stage1
			}
			remove_province_modifier = summerhall_stage1
			add_province_modifier = {
				name = summerhall_stage2
				duration = -1
			}
			c_summerhall = {
				location = {
					province_event = {
						id = iron_throne.16
						days = 750
						random = 250
					}
				}
			}
		}
		else_if = {
			limit = {
				has_province_modifier = summerhall_stage2
			}
			remove_province_modifier = summerhall_stage2
			add_province_modifier = {
				name = summerhall_stage3
				duration = -1
			}
			c_summerhall = {
				location = {
					province_event = {
						id = iron_throne.16
						days = 750
						random = 250
					}
					province_event = {
						id = unoccupied.12 #dragon egg
						days = 250
						random = 75
					}
				}
			}
		}
		else_if = {
			limit = {
				has_province_modifier = summerhall_stage3
			}
			remove_province_modifier = summerhall_stage3
			add_province_modifier = {
				name = summerhall_complete
				duration = -1
			}
		}
	}
	option = {
		name = EVTOPTAiron_throne.16
		hidden_tooltip = {
			narrative_event = {
				id = iron_throne.17
				days = 25
				random = 15
			}
		}
	}
}

is_slavery_legal = {
	OR = {
		has_law = slavery_2
		has_law = slavery_3
	}
}

holding_can_be_slave_looted = {
	#Looted holding conditions
	holder_scope = {
		NOT = {
			is_liege_or_above = ROOT
		}
		NOT = {
			same_realm = ROOT
		}
	}
	OR = {
		holding_type = castle
		holding_type = city
	}
	base_value_3_trigger = yes	#Must be at least level 3
	OR = {
		#If a sub holding, must be at least level 5
		is_capital = yes
		base_value_5_trigger = yes
	}
}

holding_can_be_slave_raided = {
	#Looted holding conditions
	holder_scope = {
		NOT = {
			is_liege_or_above = ROOT
		}
		OR = {
			NOT = {
				same_realm = ROOT
			}
			war_with = ROOT
			any_liege = {
				war_with = ROOT
			}
		}
	}
	OR = {
		holding_type = castle
		holding_type = city
	}
	base_value_3_trigger = yes	#Must be at least level 3
	OR = {
		#If a sub holding, must be at least level 5
		is_capital = yes
		base_value_5_trigger = yes
	}
}

is_full_independent_trigger = {
	independent = yes
	OR = {
		tier = EMPEROR
		primary_title = {
			NOT = {
				check_variable = {
					which = "de_facto_empire"
					value = 1
				}
			}
		}
	}
}

is_pro_slavery_trigger = {
	NOT = {
		trait = nightswatch
		trait = liberator
		has_religion_feature = religion_feature_anti_slavery
		has_dynasty_flag = liberator_dynasty
	}
	OR = {
		religion = pirate
		religion = corsair
		religion = lady_of_the_waves
		AND = {
			religion = lady_of_the_waves_reformed
			OR = {
				NOT = {
					has_dlc = "Holy Fury"
				}
				NOT = {
					has_religion_feature = religion_peaceful
				}
			}
		}
		OR = {
			NOT = {
				has_dlc = "Holy Fury"
			}
			NOT = {
				has_religion_feature = religion_feature_pro_slavery
			}
		}
		culture_group = ghiscari_group
		culture_group = valyrian
		AND = {
			culture_group = free_cities
			NOT = {
				culture = braavosi
			}
			NOT = {
				culture = lorathi
			}
		}
		culture = dothraki
		culture_group = yi_ti_group
		culture = lengi
		culture = qartheen
		culture = qartheen_lowborn
		culture_group = moraqi_group
		culture_group = asshai_group
		culture_group = nghai_group
	}
}

can_build_slavery_building = {
	#Used for buildings
	#ROOT is province, FROM is owner
	FROM = {
		NOT = {
			has_law = slavery_0
			top_liege = {
				has_law = slavery_0
			}
		}
	}
}

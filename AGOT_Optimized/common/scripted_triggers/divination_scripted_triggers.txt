can_recive_dragon_dreams = {
    is_ruler = yes
    OR = {
        trait = dragon_dreams
        AND = {
            any_courtier = {
                trait = dragon_dreams
                opinion = {
                    who = ROOT
                    value = 0
                }
                is_incapable = no
                prisoner = no
                NOT = {
                    is_inaccessible_trigger = true
                }
                NOT = {
                    character = ROOT
                }
            }
        }
    }
}

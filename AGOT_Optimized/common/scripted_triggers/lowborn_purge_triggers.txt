is_purgeable_trigger_2 = {
	trigger_if = {
		limit = {
			is_married = yes
		}
		is_purgeable_trigger = yes
		any_spouse = {
			is_purgeable_trigger = yes
		}
	}
	trigger_else = {
		limit = {
			ai = yes
		}
		is_purgeable_trigger = yes
	}
}

is_purgeable_trigger = {
	OR = {
		has_character_flag = purge
		AND = {
			ROOT = { 
				capital_scope = { 
					NOT = {
						region = world_jogo_nhai_plains
					}
				} 
			}
			OR = {
				culture = jogos_nhai
				culture = nghai
			}
		}
		AND = {
			ROOT = {
				NOT = {
					culture = white_walker
				}
			}
			culture = white_walker
		}
		NOR = {
			has_character_flag = no_purge
			is_landed = yes
			historical = yes
			has_job_title = yes
			has_minor_title = yes
			is_close_relative = ROOT
			is_lover = ROOT
			prisoner = yes
			NOT = {
				dynasty = none
			}
			is_of_bloodline = yes
			trait = maester
			trait = archmaester
			trait = kingsguard
			NOT_mythical_creature_trigger = no
			trait = favored_by_rhllor
			trait = chosen_by_rhllor
			trait = resurrected_by_rhllor
			trait = dragon_rider
			trait = immortal
			has_character_flag = special_marshal
			has_character_flag = eternal_life_mystic
			has_character_flag = flag_hired_skinchanger
			has_character_modifier = elder_long_life_modifier
		}
	}
}

			has_bloodline_flag = blood_of_azor_ahai

		}
	}
